cmake_minimum_required(VERSION 3.22)
project(crubit_library_path LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(../../test_header.cmake)

# The caller's library search path must survive Crubit's environment setup.
if(CMAKE_HOST_WIN32)
    set(library_path_variable PATH)
    set(path_separator ";")
elseif(CMAKE_HOST_APPLE)
    set(library_path_variable DYLD_LIBRARY_PATH)
    set(path_separator ":")
else()
    set(library_path_variable LD_LIBRARY_PATH)
    set(path_separator ":")
endif()
set(library_dir "${CMAKE_CURRENT_BINARY_DIR}/library path")
if(NOT "$ENV{${library_path_variable}}" STREQUAL "")
    set(ENV{${library_path_variable}} "${library_dir}${path_separator}$ENV{${library_path_variable}}")
else()
    set(ENV{${library_path_variable}} "${library_dir}")
endif()

corrosion_experimental_crubit(library_path)
corrosion_import_crate(MANIFEST_PATH Cargo.toml)
corrosion_set_env_vars(library_path "CORROSION_TEST_LIBRARY_DIR=${library_dir}")

add_executable(main main.cpp)
target_link_libraries(main PRIVATE library_path)
